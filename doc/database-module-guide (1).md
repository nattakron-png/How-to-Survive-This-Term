# คู่มือโมดูลฐานข้อมูล, Entity และ Auth (คนที่ 1: คณิศร)

คณิศรเป็นเจ้าของ schema ทั้งหมด ทั้ง Migration ทุกเวอร์ชัน, Entity, Repository พื้นฐาน, ตารางเกม และระบบล็อกอินของ Admin งานของเพื่อนทุกคนต่อจากงานชุดนี้ จึงควร merge ให้เร็วและแบ่งเป็น PR เล็ก ๆ

## 1. Migration V9: เกมและรูปแบบการแข่ง

**ห้ามแก้ V1–V8** เพราะ Flyway เก็บ checksum ไว้ ถ้าไฟล์เปลี่ยน แอปจะไม่ยอมเริ่ม

`src/main/resources/db/migration/V9__add_games_and_tournament_format.sql`

```sql
-- เกมที่จัดแข่ง
CREATE TABLE games (
    id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    code        VARCHAR(30)  NOT NULL UNIQUE,
    name        VARCHAR(100) NOT NULL,
    min_players INT          CHECK (min_players > 0),
    logo_url    VARCHAR(500),
    is_active   BOOLEAN      NOT NULL DEFAULT TRUE,
    created_at  TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- seed เฉพาะเกมที่ทีมยืนยันแล้ว min_players เป็นค่าที่เสนอ ให้ทีมยืนยันก่อน merge
INSERT INTO games (code, name, min_players) VALUES
    ('ROV',           'ROV',           5),
    ('FREE_FIRE',     'Free Fire',     4),
    ('VALORANT',      'Valorant',      5),
    ('FIGHTING_GAME', 'Fighting Game', 1);

-- ทีมหนึ่งเล่นเกมเดียว
ALTER TABLE teams
    ADD COLUMN game_id  BIGINT REFERENCES games (id) ON DELETE RESTRICT,
    ADD COLUMN logo_url VARCHAR(500);
CREATE INDEX idx_teams_game ON teams (game_id);

-- รายการแข่ง: เกม, รูปแบบ และโลโก้
ALTER TABLE tournaments
    ADD COLUMN game_id  BIGINT REFERENCES games (id) ON DELETE RESTRICT,
    ADD COLUMN format   VARCHAR(30) NOT NULL DEFAULT 'SINGLE_ELIMINATION'
                        CHECK (format IN ('SINGLE_ELIMINATION', 'POINTS')),
    ADD COLUMN logo_url VARCHAR(500);
CREATE INDEX idx_tournaments_game_start ON tournaments (game_id, start_date);
```

เหตุผลของการตัดสินใจ (เขียนลง `data-dictionary.md` ด้วย):

- **`game_id` ยังเป็น nullable:** ตามสเปกใน PROGRESS ข้อมูลเก่าในเครื่องเพื่อนจะได้ migrate ผ่าน ไม่ต้องเดาเกมให้ ฝั่ง Service บังคับให้ข้อมูลใหม่ต้องมีเกม เมื่อทุกคนล้างข้อมูลทดสอบแล้ว ค่อยเพิ่ม `SET NOT NULL` ใน Migration ถัดไป
- **`format` อยู่ที่รายการ ไม่ได้อยู่ที่เกม:** ยืดหยุ่นกว่า เช่น วันหลังอยากจัด ROV แบบอื่น Service ต้องตรวจว่าเกม Free Fire ใช้ `POINTS` และเกมอื่นใช้ `SINGLE_ELIMINATION`
- **ใช้ `code` แทนการอ้างชื่อเกม:** โค้ดเช็กเกมด้วย `FREE_FIRE` ได้ ถ้าวันหลังเปลี่ยนชื่อที่แสดงก็ไม่กระทบโค้ด

## 2. Migration V10: ผลและเกณฑ์คะแนน Free Fire

`src/main/resources/db/migration/V10__add_free_fire_points.sql`

```sql
-- เกณฑ์ของรายการแบบเก็บคะแนน
ALTER TABLE tournaments
    ADD COLUMN total_games     SMALLINT CHECK (total_games > 0),
    ADD COLUMN points_per_kill SMALLINT NOT NULL DEFAULT 1 CHECK (points_per_kill >= 0);

-- แบบเก็บคะแนนต้องมีจำนวนเกม แบบแพ้คัดออกต้องไม่มี
ALTER TABLE tournaments
    ADD CONSTRAINT chk_tournaments_total_games
    CHECK ((format = 'POINTS') = (total_games IS NOT NULL));

-- คะแนนของแต่ละอันดับ ตั้งได้ต่อรายการ
CREATE TABLE tournament_placement_points (
    tournament_id BIGINT   NOT NULL REFERENCES tournaments (id) ON DELETE CASCADE,
    placement     SMALLINT NOT NULL CHECK (placement >= 1),
    points        SMALLINT NOT NULL CHECK (points >= 0),
    PRIMARY KEY (tournament_id, placement)
);

-- เกมย่อยของรายการ เช่น เกมที่ 1–10
CREATE TABLE free_fire_games (
    id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    tournament_id BIGINT      NOT NULL REFERENCES tournaments (id) ON DELETE CASCADE,
    game_number   SMALLINT    NOT NULL CHECK (game_number >= 1),
    scheduled_at  TIMESTAMP,
    status        VARCHAR(30) NOT NULL DEFAULT 'SCHEDULED'
                  CHECK (status IN ('SCHEDULED', 'COMPLETED')),
    created_at    TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uq_ff_games_number UNIQUE (tournament_id, game_number),
    CONSTRAINT uq_ff_games_id_tournament UNIQUE (id, tournament_id)   -- เป้าหมายของ composite FK ด้านล่าง
);
CREATE INDEX idx_ff_games_scheduled ON free_fire_games (scheduled_at);

-- ผลดิบของแต่ละทีมในแต่ละเกม ไม่เก็บคะแนน คำนวณจากเกณฑ์ของรายการ
CREATE TABLE free_fire_game_results (
    id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    game_id       BIGINT    NOT NULL,
    tournament_id BIGINT    NOT NULL,
    team_id       BIGINT    NOT NULL,
    placement     SMALLINT  NOT NULL CHECK (placement >= 1),
    kills         SMALLINT  NOT NULL CHECK (kills >= 0),
    created_at    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_ffr_game FOREIGN KEY (game_id, tournament_id)
        REFERENCES free_fire_games (id, tournament_id) ON DELETE CASCADE,
    CONSTRAINT fk_ffr_participant FOREIGN KEY (tournament_id, team_id)
        REFERENCES tournament_teams (tournament_id, team_id) ON DELETE RESTRICT,
    CONSTRAINT uq_ffr_team UNIQUE (game_id, team_id),
    CONSTRAINT uq_ffr_placement UNIQUE (game_id, placement)
);
CREATE INDEX idx_ffr_tournament_team ON free_fire_game_results (tournament_id, team_id);
```

จุดที่ควรอธิบายได้ตอนนำเสนอ:

- **Composite FK ไป `tournament_teams`:** ฐานข้อมูลกันไม่ให้บันทึกผลของทีมที่ไม่ได้อยู่ในรายการนั้นเลย ไม่ต้องพึ่ง Service อย่างเดียว
- **Composite FK `(game_id, tournament_id)`:** กันผลที่ `tournament_id` ไม่ตรงกับรายการของเกมย่อย
- **ไม่เก็บคะแนนในตารางผล:** คะแนน = คะแนนอันดับจาก `tournament_placement_points` + `kills × points_per_kill` ถ้าแก้เกณฑ์ คะแนนคำนวณใหม่ทันที ต้องล็อกไม่ให้แก้เกณฑ์หลังรายการเริ่มแข่ง (กฎใน Service)
- **CHECK ข้ามคอลัมน์ `chk_tournaments_total_games`:** บังคับระดับแถวว่าแบบเก็บคะแนนต้องมี `total_games`
- **ตัด `free_fire_scoring_rules` และ `booyah_bonus` ออกจากสเปกเดิม:** เกณฑ์มีแค่ค่าเดียวต่อรายการ (`points_per_kill`) จึงใส่ในตาราง `tournaments` ได้ ส่วนโบนัส Booyah ซ้ำกับคะแนนอันดับ 1 อยู่แล้ว

**ไม่ seed คะแนนอันดับใน Migration** ให้ `TournamentService` ใส่ค่าเริ่มต้นตามมาตรฐาน FFWS (ดูหัวข้อ 6) ตอนสร้างรายการ Free Fire แล้ว Admin แก้ได้ก่อนเริ่มแข่ง

## 3. Entity ที่ต้องเพิ่มหรือแก้

`ddl-auto=validate` จะตรวจว่า Entity ตรงกับตารางทุกคอลัมน์ ถ้าไม่ตรง แอปไม่เริ่ม จึงต้องแก้ Entity ใน PR เดียวกับ Migration

| Entity | การเปลี่ยนแปลง | Migration |
| --- | --- | --- |
| `Game` (ใหม่) | `id`, `code`, `name`, `minPlayers`, `logoUrl`, `isActive`, `createdAt` | V9 |
| `Team` | เพิ่ม `@ManyToOne(fetch = LAZY) Game game`, `logoUrl` | V9 |
| `Tournament` | เพิ่ม `Game game`, `@Enumerated(STRING) TournamentFormat format`, `logoUrl` | V9 |
| `Tournament` | เพิ่ม `Short totalGames`, `Short pointsPerKill` | V10 |
| `TournamentPlacementPoint` (ใหม่) | `@EmbeddedId` (`tournamentId`, `placement`) แบบเดียวกับ `TournamentTeamId`, `points` | V10 |
| `FreeFireGame` (ใหม่) | `tournament`, `gameNumber`, `scheduledAt`, `status`, `createdAt` | V10 |
| `FreeFireGameResult` (ใหม่) | `game`, `tournament`, `team`, `placement`, `kills`, `createdAt` | V10 |

Enum ที่สร้างใน `domain/enums/` (เพื่อนใช้ร่วมกัน สร้างครั้งเดียวที่นี่):

- `TournamentFormat`: `SINGLE_ELIMINATION`, `POINTS` (คนที่ 4 ใช้เลือก Strategy)
- `TournamentStatus`: `UPCOMING`, `ONGOING`, `COMPLETED`
- `MatchStatus`: ณัฐกรสร้างไว้แล้ว (`PENDING`, `SCHEDULED`, `COMPLETED`) ใช้ตัวเดิม

**เปลี่ยน `status` ใน `Match` และ `Tournament` จาก `String` เป็น Enum:** แนะนำให้ทำ แต่ต้องบอกณัฐกรกับคนที่ 4 ก่อน เพราะโค้ดของสองคนนี้เรียก `setStatus(MatchStatus.COMPLETED.name())` และ `setStatus("ONGOING")` อยู่ ถ้าเปลี่ยนแล้วต้องแก้ไปพร้อมกัน

`FreeFireGameResult` มี `tournament` สองทาง (ตรงและผ่าน `game`) ให้ map คอลัมน์ `tournament_id` แค่ครั้งเดียว อีกตัวใส่ `insertable = false, updatable = false` ไม่อย่างนั้น Hibernate จะฟ้องว่ามีคอลัมน์ซ้ำ

## 4. Game API

ใช้ Team API ของวัชรพลเป็นแบบ ทั้ง package, Constructor Injection, `record` DTO, Mapper และ Exception

| Method | Endpoint | สิทธิ์ | Status |
| --- | --- | --- | --- |
| GET | `/api/v1/games` | ทุกคน (ตัวกรองเกมของหน้าผู้ชม) | 200 |
| GET | `/api/v1/games/{id}` | ทุกคน | 200 / 404 |
| POST | `/api/v1/games` | Admin | 201 / 400 / 409 |
| PUT | `/api/v1/games/{id}` | Admin | 200 / 400 / 404 / 409 |
| DELETE | `/api/v1/games/{id}` | Admin | 204 / 404 / 409 |

กฎใน `GameServiceImpl`:

- `code` ซ้ำ → `BusinessException` (409)
- `GameRequest`: `code` ต้อง `@NotBlank` และ `@Pattern(regexp = "[A-Z_]+")`, `name` ต้อง `@NotBlank`, `minPlayers` ต้อง `@Min(1)` → 400
- ลบเกมที่มีทีมหรือรายการใช้อยู่ → 409 ให้ตรวจก่อนลบด้วย `teamRepository.existsByGameId()` และ `tournamentRepository.existsByGameId()` อย่าปล่อยให้ FK RESTRICT โยน 500 ออกไป
- `GET /api/v1/games` คืนเฉพาะเกมที่ `is_active = true` สำหรับผู้ชม

Unit Test อย่างน้อย: สร้างสำเร็จ, code ซ้ำ, แก้เกมที่ไม่มี และลบเกมที่ยังมีทีมใช้อยู่

## 5. Auth สำหรับ Admin

ผู้ชมไม่ต้องล็อกอิน มีแค่ Admin ทำเป็นอย่างสุดท้าย หลังทีมเลือก Frontend แล้ว เพราะวิธีล็อกอินต่างกัน (Thymeleaf ใช้ session + form login ส่วน React ใช้ token)

- **กติกาสิทธิ์:** `GET /api/v1/**` เปิดให้ทุกคน ส่วน `POST`, `PUT`, `DELETE` ต้องเป็น `ADMIN` และ `/swagger-ui/**`, `/v3/api-docs/**`, `/actuator/health` เปิดให้ทุกคน
- **รหัสผ่าน:** เก็บด้วย `BCryptPasswordEncoder` ห้ามเก็บรหัสผ่านจริงในตาราง `users`
- **บัญชี Admin แรก:** อย่าใส่รหัสผ่านใน Migration ให้สร้างตอนแอปเริ่มด้วย `CommandLineRunner` ที่อ่าน `ADMIN_USERNAME` และ `ADMIN_PASSWORD` จาก Environment Variable ถ้ายังไม่มีบัญชีค่อยสร้าง และเพิ่มสองค่านี้ใน `.env.example`
- **ระหว่างพัฒนา:** เมื่อเปิด Security แล้ว เทสต์ของเพื่อนที่เรียก `POST` จะได้ 401 ต้องเพิ่ม `@WithMockUser(roles = "ADMIN")` หรือ `spring-security-test` บอกทีมก่อน merge

## 6. การเก็บคะแนน Free Fire (ข้อมูลอ้างอิง)

รายการแข่ง Free Fire ระดับอาชีพ (FFWS) ใช้ระบบเดียวกันมาหลายปี เหมาะจะเป็นค่าเริ่มต้นของระบบ:

- ทีมละ 4 คน แข่งพร้อมกันเกมละ 12 ทีม
- **คะแนนอันดับ:** 1st 12, 2nd 9, 3rd 8, 4th 7, 5th 6, 6th 5, 7th 4, 8th 3, 9th 2, 10th 1, 11th–12th 0
- **คะแนน kill:** 1 คะแนนต่อ 1 kill
- **คะแนนต่อเกม** = คะแนนอันดับ + kill และคะแนนรวมคือผลรวมทุกเกม
- **คะแนนเท่ากัน ตัดสินตามลำดับ:** จำนวน Booyah (อันดับ 1) รวม → kill รวม → อันดับในเกมล่าสุด

ในระบบนี้: คะแนนอันดับเก็บใน `tournament_placement_points` (ใส่ค่าข้างบนเป็นค่าเริ่มต้นตอนสร้างรายการ), คะแนน kill เก็บใน `tournaments.points_per_kill` (ค่าเริ่มต้น 1) และ Booyah นับจาก `placement = 1` ไม่ต้องมีคอลัมน์แยก

รายการ FFWS รอบชิงบางรายการใช้กติกาเสริมชื่อ Champion Rush (ทีมที่คะแนนถึงเกณฑ์แล้วได้ Booyah ก่อนเป็นแชมป์ทันที) ระบบนี้ยังไม่ต้องรองรับ

## 7. ลำดับ PR และ Commit

แยกเป็น PR เล็ก เพื่อนจะได้ใช้งานเร็วและรีวิวง่าย

**PR 1: เก็บกวาด** (merge ก่อนทุกอย่าง)
`chore: remove tracked .DS_Store` → `refactor: move project into code folder` → `refactor: remove inverse one-to-one from Match`

**PR 2: V9 + Game** (Reviewer: วัชรพล เพราะแก้ `Team`)
`feat(db): add games and tournament format` → `feat(entity): add Game entity and enums` → `feat(entity): map game and logo on team and tournament` → `feat: add GameRepository` → `feat: add game DTO and mapper` → `feat: add GameService` → `feat: add GameController` → `test: add unit tests for GameService`

**PR 3: V10** (Reviewer: ณัฐกร เพราะใช้ตารางผล)
`feat(db): add free fire points tables` → `feat(entity): add placement points entity` → `feat(entity): add free fire game and result entities` → `feat: add free fire repositories` → `test: verify migrations on empty database`

**PR 4: Auth** (หลังเลือก Frontend)
`feat: add spring security config` → `feat: create admin account from environment` → `feat: add auth endpoints` → `test: secure write endpoints`

ทำวันละ 2–3 commit จะได้ commit กระจายตามใบงาน
